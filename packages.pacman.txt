# Arch equivalent of packages.dnf.txt - official repos only (AUR is in
# packages.aur.txt, laptop GPU stuff in packages.pacman.nvidia.txt).
#
#   grep -vE '^\s*(#|$)' packages.pacman.txt | sudo pacman -S --needed -
#
# Names were checked against archlinux.org's package API. Arch has no -devel
# split (headers ship in the main package), so every Fedora *-devel line from
# the dnf list simply disappears.

# Hyprland itself - stock extra/hyprland, NOT the Fedora tearing-fix fork.
# build.sh skips the Hyprland fork build on Arch; test stock first, and only
# bring the fork over if tearing is still broken. (hyprland-uwsm.desktop ships
# inside the hyprland package; uwsm itself is separate.)

hyprland-guiutils
uwsm
